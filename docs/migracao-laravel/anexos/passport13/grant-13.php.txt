// cliente no esquema ANTIGO (id inteiro, segredo em texto puro, password_client) sem tocar o banco
class ClienteMem extends Laravel\Passport\Client {
    public function __construct() { $this->setRawAttributes(['id' => 9999999, 'name' => 'e-cidade', 'secret' => 'segredo-de-teste', 'redirect' => 'http://localhost', 'personal_access_client' => 0, 'password_client' => 1, 'revoked' => 0, 'provider' => null], true); }

}
class ClientePessoalMem extends Laravel\Passport\Client {
    public function __construct() { $this->setRawAttributes(['id' => 9999998, 'name' => 'MatriculaOnline', 'secret' => 'segredo-pessoal', 'redirect' => '', 'personal_access_client' => 1, 'password_client' => 0, 'revoked' => 0, 'provider' => null], true); }

    public function getIncrementing() { return true; }
}
class ClientesMem extends Laravel\Passport\ClientRepository {
    public function __construct() {}
    public function personalAccessClient(string $provider): Laravel\Passport\Client { return new ClientePessoalMem(); }
    public function findActive(string|int $id): ?Laravel\Passport\Client { return (string) $id === '9999999' ? new ClienteMem() : null; }
}
class AccessMem extends Bridge\AccessTokenRepository {
    public static $vivos = [];
    public function __construct() {}
    public function persistNewAccessToken(League\OAuth2\Server\Entities\AccessTokenEntityInterface $t): void { static::$vivos[$t->getIdentifier()] = true; }
    public function revokeAccessToken(string $id): void { unset(static::$vivos[$id]); }
    public function isAccessTokenRevoked(string $id): bool { return !isset(static::$vivos[$id]); }
}
class RefreshMem extends Bridge\RefreshTokenRepository {
    public static $vivos = [];
    public function __construct() {}
    public function persistNewRefreshToken(League\OAuth2\Server\Entities\RefreshTokenEntityInterface $t): void { static::$vivos[$t->getIdentifier()] = true; }
    public function revokeRefreshToken(string $id): void { unset(static::$vivos[$id]); }
    public function isRefreshTokenRevoked(string $id): bool { return !isset(static::$vivos[$id]); }
}
